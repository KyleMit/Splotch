import { createRequire } from 'node:module';
import { mkdir, writeFile } from 'node:fs/promises';
import assert from 'node:assert/strict';
const root = '/Users/kylemit/.codex/worktrees/native-coloring-pages/Splotch';
const require = createRequire(`${root}/package.json`);
const { chromium } = require('playwright');
const sharp = require('sharp');
const out = '/private/tmp/splotch-native-coloring-pages';
await mkdir(out, { recursive: true });
const browser = await chromium.launch({ headless: true });
const context = await browser.newContext({ viewport: { width: 1024, height: 900 }, acceptDownloads: true });
const page = await context.newPage();
const errors = [];
page.on('pageerror', error => errors.push(error.message));
const paper = page.getByTestId('drawing-paper');
const paths = () => paper.locator('svg path');
const picture = () => paths().evaluateAll(items => items.map(item => ({ d: item.getAttribute('d'), stroke: item.getAttribute('stroke'), width: item.getAttribute('stroke-width') })));
async function choose(name) {
  await page.getByRole('button', { name: 'Coloring pages', exact: true }).click();
  await page.getByRole('dialog', { name: 'Coloring pages' }).waitFor();
  await page.getByRole('radio', { name, exact: true }).click();
  await page.getByRole('dialog', { name: 'Coloring pages' }).waitFor({ state: 'hidden' });
}
try {
  await page.goto('http://localhost:5310', { waitUntil: 'networkidle', timeout: 60000 });
  await paper.waitFor();
  await page.getByRole('button', { name: 'Coloring pages', exact: true }).click();
  const dialog = page.getByRole('dialog', { name: 'Coloring pages' });
  await dialog.waitFor();
  const backdrop = await dialog.evaluate(el => getComputedStyle(el.parentElement).backgroundColor);
  assert.equal(backdrop, 'rgba(23, 23, 29, 0.72)');
  for (const name of ['Blank paper', 'Sunshine', 'Garden flower', 'Little turtle']) {
    assert.equal(await page.getByRole('radio', { name, exact: true }).count(), 1);
  }
  assert.equal(await page.getByRole('radio', { name: 'Blank paper', exact: true }).getAttribute('aria-checked'), 'true');
  await dialog.screenshot({ path: `${out}/picker.png`, animations: 'disabled' });
  await page.getByRole('radio', { name: 'Garden flower', exact: true }).click();
  await dialog.waitFor({ state: 'hidden' });
  const outline = await picture();
  assert.equal(outline.length, 4);
  await page.getByRole('button', { name: 'Green paint', exact: true }).click();
  await page.getByRole('button', { name: 'Marker', exact: true }).click();
  const bounds = await paper.boundingBox();
  assert.ok(bounds);
  await page.mouse.move(bounds.x + bounds.width * .4, bounds.y + bounds.height * .5);
  await page.mouse.down();
  await page.mouse.move(bounds.x + bounds.width * .6, bounds.y + bounds.height * .5, { steps: 12 });
  await page.mouse.up();
  assert.equal(await paths().count(), 5);
  const painted = await picture();
  const ink = painted.at(-1);
  assert.equal(ink.width, '22');
  assert.notEqual(ink.stroke, outline[0].stroke);
  await page.screenshot({ path: `${out}/flower-painted.png` });
  await page.getByRole('button', { name: 'Save picture', exact: true }).click();
  await page.getByText('Picture saved on this device.', { exact: true }).waitFor();
  const saved = await page.evaluate(() => {
    const key = Object.keys(localStorage).find(key => key.startsWith('splotch-picture:'));
    return { key, text: localStorage.getItem(key) };
  });
  const snapshot = JSON.parse(saved.text);
  assert.equal(snapshot.version, 2);
  assert.equal(snapshot.pageId, 'flower');
  assert.equal(snapshot.strokes.length, 1);
  assert.equal(snapshot.strokes[0].color, 'Green');
  assert.equal(snapshot.strokes[0].brush, 'marker');
  await writeFile(`${out}/saved-picture.json`, saved.text);
  await choose('Little turtle');
  assert.equal(await paths().count(), 8);
  await page.screenshot({ path: `${out}/turtle.png` });
  await page.getByRole('button', { name: 'Undo', exact: true }).click();
  assert.deepEqual(await picture(), painted);
  await page.getByRole('button', { name: 'Clear', exact: true }).click();
  assert.deepEqual(await picture(), outline);
  await page.getByRole('button', { name: 'Undo', exact: true }).click();
  assert.deepEqual(await picture(), painted);
  await choose('Blank paper');
  assert.equal(await paths().count(), 0);
  await page.screenshot({ path: `${out}/blank.png` });
  await page.getByRole('button', { name: 'Undo', exact: true }).click();
  assert.deepEqual(await picture(), painted);
  await page.reload({ waitUntil: 'networkidle' });
  await page.getByRole('button', { name: 'Pictures', exact: true }).click();
  await page.getByRole('button', { name: /^Open picture from / }).click();
  await page.getByText('Picture opened. Undo returns to your previous picture.', { exact: true }).waitFor();
  assert.deepEqual(await picture(), painted);
  const downloadEvent = page.waitForEvent('download');
  await page.getByRole('button', { name: 'Export PNG', exact: true }).click();
  const download = await downloadEvent;
  await download.saveAs(`${out}/flower-painted.png.export.png`);
  const { data, info } = await sharp(`${out}/flower-painted.png.export.png`).removeAlpha().raw().toBuffer({ resolveWithObject: true });
  assert.equal(info.width, 1024);
  assert.equal(info.height, 768);
  const inkRgb = ink.stroke.slice(1).match(/../g).map(hex => parseInt(hex, 16));
  const outlineRgb = outline[0].stroke.slice(1).match(/../g).map(hex => parseInt(hex, 16));
  let inkPixels = 0, outlinePixels = 0;
  for (let i = 0; i < data.length; i += info.channels) {
    if (inkRgb.every((value, channel) => Math.abs(data[i + channel] - value) < 4)) inkPixels++;
    if (outlineRgb.every((value, channel) => Math.abs(data[i + channel] - value) < 4)) outlinePixels++;
  }
  assert.ok(inkPixels > 1500, `paint pixels ${inkPixels}`);
  assert.ok(outlinePixels > 5000, `outline pixels ${outlinePixels}`);
  await page.evaluate(saved => localStorage.setItem(saved.key, JSON.stringify({ ...JSON.parse(saved.text), pageId: 'corrupt-page' })), saved);
  await page.getByRole('button', { name: 'Pictures', exact: true }).click();
  await page.getByRole('button', { name: /^Open picture from / }).click();
  await page.getByText('This saved picture is not a supported drawing.', { exact: true }).waitFor();
  await page.getByRole('button', { name: 'Close', exact: true }).click();
  assert.deepEqual(await picture(), painted);
  await page.evaluate(() => {
    Storage.prototype.setItem = () => { throw new Error('Picture storage denied.'); };
  });
  await page.getByRole('button', { name: 'Save picture', exact: true }).click();
  await page.getByText('Picture storage denied.', { exact: true }).waitFor();
  assert.deepEqual(await picture(), painted);
  await choose('Sunshine');
  assert.equal(await paths().count(), 4);
  await page.screenshot({ path: `${out}/sunshine.png` });
  const phoneContext = await browser.newContext({ viewport: { width: 390, height: 844 } });
  const phone = await phoneContext.newPage();
  await phone.goto('http://localhost:5310', { waitUntil: 'networkidle' });
  await phone.getByRole('button', { name: 'Coloring pages', exact: true }).click();
  const phoneDialog = phone.getByRole('dialog', { name: 'Coloring pages' });
  await phoneDialog.waitFor();
  for (const name of ['Blank paper', 'Sunshine', 'Garden flower', 'Little turtle']) {
    const bounds = await phone.getByRole('radio', { name, exact: true }).boundingBox();
    assert.ok(bounds && bounds.width >= 52 && bounds.height >= 52);
    assert.ok(bounds.x >= 0 && bounds.x + bounds.width <= 390);
  }
  await phone.screenshot({ path: `${out}/picker-phone.png`, animations: 'disabled' });
  await phone.getByRole('button', { name: 'Keep drawing', exact: true }).click();
  await phoneDialog.waitFor({ state: 'hidden' });
  await phoneContext.close();
  assert.deepEqual(errors, []);
  const result = { passed: true, pageId: snapshot.pageId, strokeCount: snapshot.strokes.length, inkPixels, outlinePixels, png: { width: info.width, height: info.height }, pageErrors: errors, checks: ['visible named picker and all choices', 'selected page under real green marker stroke', 'page change plus undo restores page and paint', 'clear preserves selected page and undo restores paint', 'blank selection clears page and paint atomically and undo restores both', 'save, reload, reopen exact page and stroke SVG', 'PNG contains selected outline and expected green paint pixels', 'corrupt saved page refusal preserves live picture', 'denied save preserves live picture', 'phone picker fits and all page targets exceed 52px', 'modal backdrop uses the actual design-token color'], screenshots: ['picker.png', 'flower-painted.png', 'turtle.png', 'blank.png', 'sunshine.png', 'picker-phone.png'] };
  await writeFile(`${out}/result.json`, JSON.stringify(result, null, 2));
  console.log(JSON.stringify(result));
} finally {
  await browser.close();
}
