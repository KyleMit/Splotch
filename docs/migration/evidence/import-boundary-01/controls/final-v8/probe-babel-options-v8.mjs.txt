import assert from 'node:assert/strict';
import { cpSync, mkdirSync, readFileSync, realpathSync, symlinkSync, unlinkSync, writeFileSync } from 'node:fs';
import { createRequire } from 'node:module';
import { spawnSync } from 'node:child_process';
import { dirname, join } from 'node:path';
import { pathToFileURL } from 'node:url';
import { assertDeclaredCandidateImports as currentGuard } from '/Users/kylemit/.codex/worktrees/3218/Splotch/tools/migration/lib/candidate-imports.mjs';
import { assertDeclaredCandidateImports as oldGuard } from '/private/tmp/splotch-resume-01a111b0/import-boundary/final-v8/old-v7/candidate-imports.mjs';
const worktree = '/Users/kylemit/.codex/worktrees/3218/Splotch';
const root = '/private/tmp/splotch-resume-01a111b0/import-boundary';
const evidence = join(root, 'final-v8/babel-options-v2');
const runtimeName = 'splotch-owned-jsx-runtime-probe';
const require = createRequire(join(worktree, 'package.json'));
const babel = require('@babel/core');
const cases = ['default', 'empty', 'declared', 'undeclared', 'native-options', 'opaque-json', 'opaque-factory', 'json-config', 'named-plugin', 'json-named-plugin'];
const results = [];
mkdirSync(evidence, { recursive: true });
function capture(operation) { try { return { passed: true, result: operation() }; } catch (error) { return { passed: false, error: String(error), code: error.code }; } }
for (const name of cases) {
  const candidate = join(evidence, name, 'candidate');
  cpSync(join(worktree, 'experiments/native-architecture'), candidate, { recursive: true, filter: (path) => !path.includes('/node_modules') });
  symlinkSync(join(worktree, 'node_modules'), join(candidate, 'node_modules'));
  cpSync(join(root, 'jsx-pragma/node_modules'), join(dirname(candidate), 'node_modules'), { recursive: true });
  const manifestPath = join(candidate, 'package.json');
  const manifest = JSON.parse(readFileSync(manifestPath));
  if (name === 'declared') manifest.devDependencies[runtimeName] = '1.0.0';
  if (name.includes('named-plugin')) manifest.devDependencies['babel-plugin-react-compiler'] = require('babel-plugin-react-compiler/package.json').version;
  writeFileSync(manifestPath, JSON.stringify(manifest));
  const baseline = capture(() => currentGuard(candidate, manifest));
  const json = name.startsWith('json-');
  const configPath = join(candidate, json ? '.babelrc' : 'babel.config.cjs');
  const original = json ? null : readFileSync(configPath, 'utf8');
  let option = name === 'empty' ? '{}' : `{jsxImportSource:'${runtimeName}'}`;
  if (name === 'native-options') option = `{native:${option}}`;
  if (name === 'opaque-json') option = `JSON.parse('{"jsxImportSource":"${runtimeName}"}')`;
  if (name === 'opaque-factory') option = 'makeOptions()';
  let source = `module.exports=function(api){api.cache(true);return {presets:[[require.resolve('babel-preset-expo'),${option}]]};};\n`;
  if (name === 'default') source = original;
  if (name === 'opaque-factory') source = `function makeOptions(){return{jsxImportSource:'${runtimeName}'};}\n${source}`;
  if (name === 'json-config') source = JSON.stringify({ presets: [['babel-preset-expo', { jsxImportSource: runtimeName }]] });
  if (name === 'named-plugin') source = "module.exports={presets:[require.resolve('babel-preset-expo')],plugins:[[require.resolve('babel-plugin-react-compiler'),{},'owned-name']]};";
  if (name === 'json-named-plugin') source = JSON.stringify({ presets: ['babel-preset-expo'], plugins: [['babel-plugin-react-compiler', {}, 'owned-name']] });
  writeFileSync(configPath, source);
  const helper = join(candidate, 'src/OptionHelper.tsx');
  writeFileSync(helper, 'export const probe=<>option probe</>;\n');
  const old = capture(() => oldGuard(candidate, manifest));
  const current = capture(() => currentGuard(candidate, manifest));
  const options = { babelrc: false, configFile: configPath, cwd: candidate, envName: 'production', caller: { name: 'splotch-native-topology', platform: 'ios', supportsStaticESM: true } };
  const transformed = babel.transformFileSync(helper, options).code;
  const predicate = name.includes('named-plugin') ? /unsupported extra Babel tuple members/ : /unsupported Babel options/;
  const oldRejectAssertion = ['default', 'empty'].includes(name) ? null : capture(() => assert.throws(() => oldGuard(candidate, manifest), predicate));
  const runtimeSpecifier = ['default', 'empty', 'named-plugin', 'json-named-plugin'].includes(name) ? 'react/jsx-runtime' : `${runtimeName}/jsx-runtime`;
  assert.ok(transformed.includes(runtimeSpecifier));
  const resolved = realpathSync(createRequire(helper).resolve(runtimeSpecifier));
  const emitted = join(candidate, 'src/OptionHelper.emitted.mjs');
  writeFileSync(emitted, transformed);
  const executed = await import(pathToFileURL(emitted).href);
  if (runtimeSpecifier.startsWith(runtimeName)) assert.equal(executed.probe.marker, 'owned-hoisted-runtime');
  const typecheck = spawnSync(process.execPath, [join(worktree, 'node_modules/typescript/bin/tsc'), '--project', join(candidate, 'tsconfig.json'), '--noEmit'], { cwd: candidate, encoding: 'utf8', timeout: 30000 });
  unlinkSync(emitted); unlinkSync(helper);
  if (original) writeFileSync(configPath, original); else unlinkSync(configPath);
  const restored = capture(() => currentGuard(candidate, manifest));
  const actualMetro = createRequire(join(candidate, 'package.json'))('expo/metro-config').getDefaultConfig(candidate);
  const selector = require(join(dirname(actualMetro.transformer.babelTransformerPath), 'loadBabelConfig.js'));
  results.push({ name, candidate, manifest, source, baseline, old, current, transformed, runtimeSpecifier, resolved, runtimeMarker: executed.probe.marker ?? 'default-react-element', typecheck: { status: typecheck.status, stdout: typecheck.stdout, stderr: typecheck.stderr }, oldRejectAssertion, restored, restoredRootConfig: selector.resolveBabelrcName(candidate) });
  assert.equal(baseline.passed, true); assert.equal(old.passed, true); assert.equal(typecheck.status, 0); assert.equal(restored.passed, true);
  assert.equal(current.passed, ['default', 'empty'].includes(name));
  if (!current.passed) { assert.match(current.error, predicate); assert.equal(oldRejectAssertion.passed, false); assert.match(oldRejectAssertion.error, /Missing expected exception/); }
}
writeFileSync(join(evidence, 'receipt.json'), `${JSON.stringify({ sourceInputs: '../source-inputs.json', oldSourceInputs: '../../final-v7/source-inputs.json', versions: { babel: babel.version, babelPresetExpo: require('babel-preset-expo/package.json').version }, results }, null, 2)}\n`);
console.log(JSON.stringify({ evidence, results: results.map(({ name, current, runtimeSpecifier, runtimeMarker, typecheck, restored }) => ({ name, guard: current.passed, error: current.error, runtimeSpecifier, runtimeMarker, typecheck: typecheck.status, restored: restored.passed })) }));
