function capture(operation) { try { return { passed: true, result: operation() }; } catch (error) { return { passed: false, error: String(error) }; } }
function fixture(name) {
  const target = join(evidence, 'consumers', name, 'candidate');
  mkdirSync(dirname(target), { recursive: true });
  cpSync(join(worktree, 'experiments/native-architecture'), target, { recursive: true, filter: (path) => !path.includes('/node_modules') });
  symlinkSync(join(worktree, 'node_modules'), join(target, 'node_modules'));

}
const variants = JSON.parse(readFileSync(join(root, 'round3-diagnosis/receipt.json'))).results;
const configurationResults = [];
for (const { name, file, source } of variants) {
  const candidate = fixture(name);
  const manifest = JSON.parse(readFileSync(join(candidate, 'package.json')));
  const path = join(candidate, file);
  mkdirSync(dirname(path), { recursive: true });
  const original = name.startsWith('A') ? readFileSync(path, 'utf8') : null;
  const baseline = capture(() => currentGuard(candidate, manifest));
  writeFileSync(path, source);
  const old = capture(() => oldGuard(candidate, manifest));
  const current = capture(() => currentGuard(candidate, manifest));
  const options = { babelrc: false, configFile: join(candidate, 'babel.config.cjs'), cwd: candidate, envName: 'production', caller: { name: 'splotch-native-topology', platform: 'ios', supportsStaticESM: true } };
  const consumer = capture(() => name.startsWith('A') ? babel.loadPartialConfig({ ...options, filename: join(candidate, 'src/ProbeApp.tsx') }).options.plugins.map((item) => item.file?.resolved) : babel.transformFileSync(path, options).code);
  if (original) writeFileSync(path, original); else unlinkSync(path);
  const restored = capture(() => currentGuard(candidate, manifest));
  configurationResults.push({ name, source, candidate, baseline, old, current, consumer, restored });
  assert.equal(baseline.passed, true);
  assert.equal(consumer.passed, true);
  assert.equal(restored.passed, true);
  if (name.startsWith('A')) {
    assert.equal(old.passed, true);
    assert.equal(current.passed, false);
    assert.match(current.error, /unsupported configuration export alias/);
    assert.ok(consumer.result.some((path) => path?.includes('babel-plugin-react-compiler')));
  } else {
    assert.equal(old.passed, false);
    assert.match(old.error, /has no supported literal configuration export/);
    assert.equal(current.passed, true);
  }
}

  const babelOutput = babel.transformFileSync(path, options).code;

  const tsOutput = ts.transpileModule(source, { fileName: path, compilerOptions: parsed.options }).outputText;
  const typecheck = spawnSync(process.execPath, [join(worktree, 'node_modules/typescript/bin/tsc'), '--project', join(candidate, 'tsconfig.json'), '--noEmit'], { cwd: candidate, encoding: 'utf8', timeout: 30000 });
  writeFileSync(join(dirname(candidate), 'typecheck.log.txt'), typecheck.stdout + typecheck.stderr);
  writeFileSync(join(dirname(candidate), 'babel-output.mjs.txt'), babelOutput);
  writeFileSync(join(dirname(candidate), 'typescript-output.mjs.txt'), tsOutput);
  const hoistedOwner = realpathSync(createRequire(path).resolve('splotch-owned-jsx-runtime-probe/jsx-runtime'));
  unlinkSync(path);
  writeFileSync(entry, original);
  const restored = capture(() => currentGuard(candidate, manifest));
  jsxResults.push({ name, source, candidate, baseline, old, current, babelOutput, tsOutput, typecheck: { status: typecheck.status, stdout: typecheck.stdout, stderr: typecheck.stderr }, hoistedOwner, restored });
  assert.equal(baseline.passed, true);
  assert.equal(old.passed, true);
  assert.equal(typecheck.status, 0);
  assert.equal(current.passed, !proposedFence);
  if (proposedFence) assert.match(current.error, /unsupported per-file JSX import source/);
  assert.equal(restored.passed, true);
}
