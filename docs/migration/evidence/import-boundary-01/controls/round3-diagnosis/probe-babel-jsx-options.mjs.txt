import assert from 'node:assert/strict';
import { cpSync, mkdirSync, readFileSync, realpathSync, symlinkSync, unlinkSync, writeFileSync } from 'node:fs';
import { createRequire } from 'node:module';
import { spawnSync } from 'node:child_process';
import { dirname, join } from 'node:path';
import { pathToFileURL } from 'node:url';
import { assertDeclaredCandidateImports as currentGuard } from '/Users/kylemit/.codex/worktrees/3218/Splotch/tools/migration/lib/candidate-imports.mjs';
import { assertDeclaredCandidateImports as oldGuard } from '/private/tmp/splotch-resume-01a111b0/import-boundary/final-v7/old-ecdd/candidate-imports.mjs';
const worktree = '/Users/kylemit/.codex/worktrees/3218/Splotch';
const root = '/private/tmp/splotch-resume-01a111b0/import-boundary';
const evidence = join(root, 'babel-jsx-options');
const runtimeName = 'splotch-owned-jsx-runtime-probe';
const require = createRequire(join(worktree, 'package.json'));
const babel = require('@babel/core');
const results = [];
mkdirSync(evidence, { recursive: true });
function capture(operation) { try { return { passed: true, result: operation() }; } catch (error) { return { passed: false, error: String(error) }; } }
for (const name of ['default', 'declared', 'undeclared', 'native-options']) {
  const candidate = join(evidence, name, 'candidate');
  cpSync(join(worktree, 'experiments/native-architecture'), candidate, { recursive: true, filter: (path) => !path.includes('/node_modules') });
  symlinkSync(join(worktree, 'node_modules'), join(candidate, 'node_modules'));
  cpSync(join(root, 'jsx-pragma/node_modules'), join(dirname(candidate), 'node_modules'), { recursive: true });
  const manifestPath = join(candidate, 'package.json');
  const manifest = JSON.parse(readFileSync(manifestPath));
  if (name === 'declared') { manifest.devDependencies[runtimeName] = '1.0.0'; writeFileSync(manifestPath, JSON.stringify(manifest)); }
  const configPath = join(candidate, 'babel.config.cjs');
  const original = readFileSync(configPath, 'utf8');
  const option = name === 'default' ? '{}' : name === 'native-options' ? `{native:{jsxImportSource:'${runtimeName}'}}` : `{jsxImportSource:'${runtimeName}'}`;
  const source = `module.exports=function(api){api.cache(true);return {presets:[[require.resolve('babel-preset-expo'),${option}]]};};\n`;
  writeFileSync(configPath, source);
  const helper = join(candidate, 'src/OptionHelper.tsx');
  writeFileSync(helper, 'export const probe=<>option probe</>;\n');
  const old = capture(() => oldGuard(candidate, manifest));
  const current = capture(() => currentGuard(candidate, manifest));
  const transformed = babel.transformFileSync(helper, { babelrc: false, configFile: configPath, cwd: candidate, envName: 'production', caller: { name: 'splotch-native-topology', platform: 'ios', supportsStaticESM: true } }).code;
  const runtimeSpecifier = name === 'default' ? 'react/jsx-runtime' : `${runtimeName}/jsx-runtime`;
  assert.ok(transformed.includes(runtimeSpecifier));
  const resolved = realpathSync(createRequire(helper).resolve(runtimeSpecifier));
  const emitted = join(candidate, 'src/OptionHelper.emitted.mjs');
  writeFileSync(emitted, transformed);
  const executed = await import(pathToFileURL(emitted).href);
  if (name !== 'default') assert.equal(executed.probe.marker, 'owned-hoisted-runtime');
  const typecheck = spawnSync(process.execPath, [join(worktree, 'node_modules/typescript/bin/tsc'), '--project', join(candidate, 'tsconfig.json'), '--noEmit'], { cwd: candidate, encoding: 'utf8', timeout: 30000 });
  unlinkSync(emitted); unlinkSync(helper); writeFileSync(configPath, original);
  const restored = capture(() => currentGuard(candidate, manifest));
  results.push({ name, candidate, manifest, source, old, current, transformed, runtimeSpecifier, resolved, runtimeMarker: executed.probe.marker ?? 'default-react-element', typecheck: { status: typecheck.status, stdout: typecheck.stdout, stderr: typecheck.stderr }, restored });
  assert.equal(old.passed, true); assert.equal(current.passed, true); assert.equal(typecheck.status, 0); assert.equal(restored.passed, true);
}
writeFileSync(join(evidence, 'receipt.json'), `${JSON.stringify({ sourceInputs: '../final-v7/source-inputs.json', results }, null, 2)}\n`);
console.log(JSON.stringify({ evidence, results: results.map(({ name, current, runtimeSpecifier, runtimeMarker, typecheck }) => ({ name, guard: current.passed, runtimeSpecifier, runtimeMarker, typecheck: typecheck.status })) }));
