function capture(operation) { try { return { passed: true, result: operation() }; } catch (error) { return { passed: false, error: String(error) }; } }

  cpSync(join(worktree, 'experiments/native-architecture'), candidate, { recursive: true, filter: (path) => !path.includes('/node_modules') });
  symlinkSync(join(worktree, 'node_modules'), join(candidate, 'node_modules'));

  const path = join(candidate, variant.file);
  mkdirSync(join(candidate, 'src/config'), { recursive: true });
  const originalConfig = readFileSync(join(candidate, 'babel.config.cjs'), 'utf8');
  const baseline = capture(() => assertDeclaredCandidateImports(candidate, manifest));
  writeFileSync(path, variant.source);
  const guard = capture(() => assertDeclaredCandidateImports(candidate, manifest));
  const options = { babelrc: false, configFile: join(candidate, 'babel.config.cjs'), cwd: candidate, envName: 'production', caller: { name: 'splotch-native-topology', platform: 'ios', supportsStaticESM: true } };
  const consumer = capture(() => variant.consumer === 'config'

  const restored = capture(() => assertDeclaredCandidateImports(candidate, manifest));

}
