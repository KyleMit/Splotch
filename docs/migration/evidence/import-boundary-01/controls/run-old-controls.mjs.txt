import * as fs from "node:fs";
import assert from 'node:assert/strict';
import { execFileSync, spawnSync } from 'node:child_process';
import { readFileSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';

const worktree = '/Users/kylemit/.codex/worktrees/3218/Splotch';
const base = 'dc08a90abc67b53124146bf288c80de0f9ffd1dd';
const root = join(import.meta.dirname, "final-v1");
fs.mkdirSync(root, { recursive: true });
const target = join(worktree, 'tools/migration/lib/native-identity.mjs');
const current = readFileSync(target);
const old = execFileSync('git', ['show', `${base}:tools/migration/lib/native-identity.mjs`], { cwd: worktree });
const pattern = 'treats dot-prefixed bare|rejects direct and reachable|rejects undeclared (import-equals|static-template|element-resolve|import-type|JSDoc)|discovers previously unlisted|rejects malformed package expo/\\.\\./yaml|rejects Node builtin node:fs|rejects (tsconfig.json|app.json|babel.config.cjs) configuration|rejects dotfile Babel JSON';
const args = ['run', 'test:tools', '--', 'tools/migration/tests/candidate-imports.test.mjs', '-t', pattern, '--reporter', 'json', '--outputFile', join(root, 'old-control-results.json')];
let run;
try {
  writeFileSync(target, old);
  run = spawnSync('npm', args, { cwd: worktree, encoding: 'utf8', timeout: 30000 });
} finally {
  writeFileSync(target, current);
}
writeFileSync(join(root, 'old-controls.log.txt'), `${run.stdout}\n${run.stderr}`);
assert.equal(run.status, 1, 'Old source must fail the new rejecting controls');
assert.deepEqual(readFileSync(target), current, 'Final source must be restored');
const tests = JSON.parse(readFileSync(join(root, 'old-control-results.json'), 'utf8'));
const failures = tests.testResults.flatMap((suite) => suite.assertionResults.filter((test) => test.status === 'failed'));
assert.ok(failures.length > 0, 'Negative controls must execute');
assert.ok(failures.every((test) => test.failureMessages.every((message) => message.includes('expected [Function] to throw an error'))), 'Failures must come from rejecting-control assertions, not setup');
const receipt = { base, worktree, executable: 'npm', args, exitCode: run.status, signal: run.signal, failures: failures.map(({ fullName, failureMessages }) => ({ fullName, failureMessages })), finalSourceRestored: true, scope: 'New false-pass regression assertions against exact accepted old checker; old source restored after execution.' };
writeFileSync(join(root, 'old-controls.json'), `${JSON.stringify(receipt, null, 2)}\n`);
console.log(JSON.stringify({ exitCode: run.status, failingRegressionControls: failures.length, finalSourceRestored: true }));
