import * as fs from "node:fs";
import assert from 'node:assert/strict';
import { spawnSync } from 'node:child_process';
import { existsSync, readFileSync, rmSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import { pathToFileURL } from 'node:url';

const worktree = '/Users/kylemit/.codex/worktrees/3218/Splotch';
const root = join(import.meta.dirname, "final-v2");
fs.mkdirSync(root, { recursive: true });
const candidate = join(worktree, 'experiments/native-architecture');
const entry = join(candidate, 'src/index.ts');
const extra = join(candidate, 'src/Extra.ts');
const original = readFileSync(entry);
assert.equal(existsSync(extra), false, 'Process controls may not overwrite existing source');
const { assertDeclaredCandidateImports } = await import(pathToFileURL(join(worktree, 'tools/migration/lib/native-identity.mjs')));
const manifest = JSON.parse(readFileSync(join(candidate, 'package.json'), 'utf8'));
const records = [];
function run(name, command, args) {
  const result = spawnSync(command, args, { cwd: worktree, encoding: 'utf8', timeout: 30000 });
  const record = { name, command, args, exitCode: result.status, signal: result.signal, stdout: result.stdout, stderr: result.stderr, error: result.error?.message ?? null };
  records.push(record);
  writeFileSync(join(root, `${name}.log.txt`), `${result.stdout}\n${result.stderr}`);
  return record;
}
try {
  writeFileSync(entry, `${original}\nimport 'yaml';\n`);
  const direct = run('direct-real-checker', 'npm', ['run', 'check:migration:native-topology']);
  assert.equal(direct.exitCode, 1);
  assert.ok(`${direct.stdout}${direct.stderr}`.includes('src/index.ts imports undeclared yaml'));
  writeFileSync(entry, original);
  assertDeclaredCandidateImports(candidate, manifest);
  records.push({ name: 'direct-restored-real-guard', passed: true });
  writeFileSync(entry, `${original}\nimport './Extra';\n`);
  writeFileSync(extra, "import { parse } from 'yaml'; parse('x: 1');\n");
  const reachable = run('reachable-real-checker', 'npm', ['run', 'check:migration:native-topology']);
  assert.equal(reachable.exitCode, 1);
  assert.ok(`${reachable.stdout}${reachable.stderr}`.includes('src/Extra.ts imports undeclared yaml'));
  const context = run('reachable-typecheck-context', process.execPath, [join(worktree, 'node_modules/typescript/bin/tsc'), '--project', join(candidate, 'tsconfig.json'), '--noEmit']);
  assert.equal(context.exitCode, 0, 'The source-isolation rejecting control must still typecheck');
} finally {
  writeFileSync(entry, original);
  rmSync(extra, { force: true });
}
assertDeclaredCandidateImports(candidate, manifest);
records.push({ name: 'reachable-restored-real-guard', passed: true });
const restored = run('restored-candidate-typecheck', process.execPath, [join(worktree, 'node_modules/typescript/bin/tsc'), '--project', join(candidate, 'tsconfig.json'), '--noEmit']);
assert.equal(restored.exitCode, 0);
assert.deepEqual(readFileSync(entry), original);
writeFileSync(join(root, 'process-controls.json'), `${JSON.stringify({ worktree, records, allSourceRestored: true, scope: 'Actual public topology command rejects direct and reachable imports for the intended ownership reason. Hoisted tsc pass is context only, not dependency-isolation evidence.' }, null, 2)}\n`);
console.log(JSON.stringify({ directRejected: true, reachableRejected: true, mutatedTypecheckExit: 0, allSourceRestored: true }));
