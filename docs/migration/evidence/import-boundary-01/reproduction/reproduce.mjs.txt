function check(name, expectedPass) {
  let error = null;
  try { assertDeclaredCandidateImports(probe, manifest); }
  catch (cause) { error = { name: cause.name, message: cause.message }; }
  assert.equal(error === null, expectedPass, `${name}: wrong guard disposition`);
  const outcome = { name, passed: error === null, error };
  outcomes.push(outcome);
  return outcome;
}
function typecheck(name) {
  const args = [join(donor, 'node_modules/typescript/bin/tsc'), '--project', join(probe, 'tsconfig.json'), '--noEmit'];

  const outcome = { name, executable: process.execPath, args, exitCode: result.status, signal: result.signal, stdout: result.stdout, stderr: result.stderr, error: result.error?.message ?? null };
  assert.equal(result.status, 0, `${name}: typecheck setup/fixture error`);
  outcomes.push(outcome);
}
