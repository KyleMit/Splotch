  set -e  # Re-enable exit on error
  # if the pnpm config set was failing it's not supported and fall back to setting the environment variable
  # for older pnpm versions that does not support the store-dir config
  if [ "$exit_status" -ne "0" ]
  then
    major_version=$(pnpm --version | cut -d'.' -f1)
    if [ "$major_version" -gt 6 ]
    then
      # in PNPM version 7.0.0 it's not working with set store-dir only in higher pnpm versions like 7.14.x
      echo "store-dir = \"$HOME/.pnpm-store\"" >> "$HOME/.npmrc"
    else
      # older versions like 6 and below use the environment variable
      export PNPM_STORE_DIR="$HOME/.pnpm-store"
    fi
  fi

  restore_node_modules "pnpm"
  echo "Installing npm packages using pnpm version $(pnpm --version)"
  if pnpm install ${PNPM_FLAGS:+$PNPM_FLAGS}
  then
    echo "npm packages installed using pnpm"
  else
    echo "Error during pnpm install"
    exit 1
  fi

  export PATH=$PATH:$(pnpm bin)
}
