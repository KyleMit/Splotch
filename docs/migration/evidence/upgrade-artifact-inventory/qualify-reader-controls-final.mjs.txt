import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { execFileSync } from 'node:child_process';
import { mkdirSync, readFileSync, rmSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import { readAabVersion, readIpaVersion } from '/Users/kylemit/Code/Splotch/tools/release/lib/artifact-version.mjs';

const REPO = '/Users/kylemit/Code/Splotch';
const ROOT = '/private/tmp/splotch-resume-01a111b0/continuity-intake';
const CONTROL_ROOT = join(ROOT, 'reader-controls-final');
const SHA = (bytes) => createHash('sha256').update(bytes).digest('hex');
const ZIP_END_SIGNATURE = Buffer.from([0x50, 0x4b, 0x05, 0x06]);
const ZIP_END_MIN_BYTES = 22;
const ZIP_COMMENT_MAX_BYTES = 65_535;
const ZIP_SIGNATURE_BYTES = 4;
const EXPECTED_FAILURE = 'not a zip archive (no end-of-central-directory record)';
const ownerPaths = ['tools/release/lib/artifact-version.mjs', 'tools/release/lib/zip.mjs'];
const owners = () => Object.fromEntries(ownerPaths.map((path) => [path, SHA(readFileSync(join(REPO, path)))]));
const sourceBefore = owners();
const git = (...args) => execFileSync('git', ['-C', REPO, ...args], { encoding: 'utf8' }).trim();
const gitBefore = { head: git('rev-parse', 'HEAD'), status: git('status', '--porcelain=v1') };
const qualificationBytes = readFileSync(join(ROOT, 'supported-artifact-qualification.json'));
const row = JSON.parse(qualificationBytes).results.find((entry) => entry.tag === 'v1.4.0');
assert.equal(row.assets.length, 2);
mkdirSync(CONTROL_ROOT);
const rows = [];
for (const asset of row.assets) {
  const sourcePath = join(ROOT, 'artifacts/v1.4.0', asset.name);
  const bytes = readFileSync(sourcePath);
  assert.equal(bytes.length, asset.bytes);
  assert.equal(SHA(bytes), asset.sha256);
  const readVersion = asset.name.endsWith('.aab') ? readAabVersion : readIpaVersion;
  const positive = readVersion(sourcePath);
  assert.deepEqual(positive, asset.actualEmbeddedVersion);
  const changed = Buffer.from(bytes);
  const endOffset = changed.lastIndexOf(ZIP_END_SIGNATURE);
  assert.ok(endOffset >= Math.max(0, bytes.length - ZIP_END_MIN_BYTES - ZIP_COMMENT_MAX_BYTES));
  changed.fill(0, endOffset, endOffset + ZIP_SIGNATURE_BYTES);
  const copyPath = join(CONTROL_ROOT, asset.name);
  writeFileSync(copyPath, changed, { flag: 'wx' });
  let observedFailure;
  assert.throws(() => readVersion(copyPath), (error) => {
    observedFailure = error.message;
    return error instanceof Error && error.message === EXPECTED_FAILURE;
  });
  writeFileSync(copyPath, bytes);
  const restored = readVersion(copyPath);
  assert.deepEqual(restored, positive);
  assert.equal(SHA(readFileSync(copyPath)), asset.sha256);
  assert.equal(SHA(readFileSync(sourcePath)), asset.sha256);
  rows.push({ assetId: asset.assetId, name: asset.name, bytes: bytes.length, sha256: asset.sha256, actualPositive: positive, changedEndOffset: endOffset, changedSignatureBytes: ZIP_SIGNATURE_BYTES, changedCopySha256: SHA(changed), actualFailure: observedFailure, actualRestoredPositive: restored, sourceUnchanged: true });
  rmSync(copyPath);
}
const sourceAfter = owners();
const gitAfter = { head: git('rev-parse', 'HEAD'), status: git('status', '--porcelain=v1') };
assert.deepEqual(sourceBefore, sourceAfter);
assert.deepEqual(gitBefore, gitAfter);
const receipt = { observedAtUtc: new Date().toISOString(), nodeVersion: process.version, producerSha256: SHA(readFileSync(new URL(import.meta.url))), reasonForInvocation: 'Original two-control receipt retained, but its inline producer was not saved as a standalone technical input; this invocation freezes exact producer/owner identities for portable verification', predicate: 'Actual v1.4 AAB/IPA version positives, one owned EOCD signature mutation refuses for exact invalid-ZIP reason, restored exact bytes return the same versions', inputs: { qualificationSha256: SHA(qualificationBytes) }, gitBefore, gitAfter, sourceBefore, sourceAfter, rows, claimLimits: ['Production readers unchanged; no extraction, signing verification, app installation/execution or upgrade acceptance', 'Negative control qualifies invalid-ZIP refusal only, not every protobuf/plist/trust-boundary behavior', 'Real tag/embedded-version mismatches remain findings, not reader rejecting controls'] };
writeFileSync(join(ROOT, 'reader-controls-final.json'), JSON.stringify(receipt, null, 2) + '\n');
process.stdout.write(JSON.stringify({ passed: rows.length, restored: rows.length, sourceUnchanged: true }) + '\n');
