// Metro bundles an entry for Node the way Expo Router's ssrLoadModule does: Babel folds
// `typeof window` to 'undefined', so every guard takes its server branch.
export async function bundleForNode(entryFile, customResolverOptions = {}) {
  const { server, bundleRequest } = await expoBundleRequestFactory()(NATIVE_APP_DIR, {
    entryFile,
    platform: 'web',
    dev: false,
    minify: false,
    resetCache: false,
  });
  try {
    const { code } = await server.build({
      ...bundleRequest,
      customTransformOptions: { ...bundleRequest.customTransformOptions, environment: 'node' },
      customResolverOptions: {
        ...bundleRequest.customResolverOptions,
        ...customResolverOptions,
        environment: 'node',
      },
    });
    return asCommonJsModule(code);
  } finally {
    await server.end();
  }
}

// Live ink renders no React (ADR-0178). Each packet paints its samples and hands the view a new
// picture as it arrives, unless it reached JavaScript too soon after the last hand-off to be
// shown before the next one (handoffCadence.ts). A contact that ends paints at once. A cohort
// that finished and was not committed after all leaves the layer by the frame after its commit
// would have rendered.
function useLiveFrames(
  scene: InkScene,
  live: LiveInkFeed,
  view: { current: SkiaPictureView | null },
  latest: {
    current: {
      strokes: readonly Stroke[];
      input: SceneInput | null;
      onError: (error: unknown) => void;
    };
  }
) {
  useEffect(() => {
    let check: number | null = null;
    // A fault detaches the input, which settles the feed again: report it once.
    let failed = false;
    // Whether it handed the view new ink.
    const paint = () => {
      const source = live.source();
      if (!source || failed) return false;
      try {
        scene.flush(source);
        const shown = present(scene, view.current);
        if (check === null && scene.orphaned(latest.current.strokes))
          check = requestAnimationFrame(() => {
            check = null;
            const { strokes, input } = latest.current;
            if (!input || !scene.orphaned(strokes)) return;
            scene.rebuild(input, source.live());
            present(scene, view.current);
          });
        return shown;
      } catch (error) {
        failed = true;
        latest.current.onError(error);
        return false;
      }
    };
    const cadence = createHandoffCadence(FRAME_CLOCK, paint);
    const unsubscribe = live.subscribe((event) => {
      if (event === 'sample') cadence.sample();
      else cadence.settle();
    });
    return () => {
      unsubscribe();
      cadence.dispose();
      if (check !== null) cancelAnimationFrame(check);
    };
  }, [scene, live, view, latest]);
}

// SkiaPictureView hands its `picture` prop to the native view through Skia's JSI view API in its
// own componentDidUpdate. The ink layer calls the same setter on each display frame, so a new
// frame of ink renders no React; the view is never given a `picture` prop to overwrite it with.
// The native view keeps its own reference to the picture it holds, so the caller may release the
// one before as soon as this returns. That relies on react-native-skia's picture renderer taking
// one lock to replace its picture on this thread and to copy it on its render thread: with nothing
// ordering the two, dropping a replaced picture's last reference can free it mid-draw ("Pure
// virtual function called!" on Android). native-picture-lifetime.test.mjs reads the renderer.
export function showPicture(view: SkiaPictureView, picture: SkPicture) {
  const api: unknown = Reflect.get(globalThis, 'SkiaViewApi');
  if (!isViewApi(api)) throw new Error('The drawing paper could not be shown.');
  api.setJsiProperty(view.nativeId, 'picture', picture);
}
// Rebuilds only the React Native app's JavaScript and puts it into a Release build that a full

## Checking a JS-only change without a native build

`swap-js-bundle.mjs` puts freshly built JavaScript into a Release build that a full Gradle or Xcode
build already produced, then installs that copy. A change to TS or TSX reaches the device in about
ten seconds instead of a native rebuild, and the device stays on a Release build, so timing checks
still hold (issue 3083). The base is the `art.splotch.app.dev` app: a plain Release build carries
that ID and installs beside the store copy.

```sh
# Android: the base APK from `npm run android:apk`, which signs with the local debug key;
# signing again with that key lets `adb install -r` keep the app's data
node tools/device-parity/swap-js-bundle.mjs --platform=android --base=<app-release.apk> \
  --bundle=art.splotch.app.dev --keystore=$HOME/.android/debug.keystore \
  --key-alias=androiddebugkey --ks-pass=pass:android --out=<new dir> --device=<serial>
# iOS: the base .app from `xcodebuild … -configuration Release`, signed again with the identity
# that signed it
node tools/device-parity/swap-js-bundle.mjs --platform=ios --base=<Splotch.app> \
  --bundle=art.splotch.app.dev --identity="Apple Development" --out=<new dir> --device=<udid>
```

Relaunch the app afterwards. `--ks-pass` and `--key-pass` take apksigner's forms (`env:NAME`,
`file:PATH`, `pass:VALUE`); without `--device` the tool only writes the swapped build into `--out`.

* **It builds what the native build builds.** The bundle step and Hermes flags are the ones React
  Native's Gradle task and Xcode script pass for Release (`lib/js-bundle.mjs`), and
  `tests/js-bundle.test.mjs` fails when React Native changes them. At an unchanged commit the
  Android bundle is byte for byte the one Gradle packaged. The iOS one differs only in the path
  Hermes records for its source. On iOS the bundle step sources the app's `.xcode.env` files, as
  Xcode's bundle phase does; on Android it uses the calling environment, as a Gradle build started
  from it does, so an `EXPO_PUBLIC_*` value applies either way. The tool prints whether the
  JavaScript differs from the base build's, which catches a forgotten save.
* **Native code and Android resources come from the base build.** A Kotlin, Swift or native config
  change needs a full build. On Android, an image or sound the base APK lacks is refused, because
  React Native compiles those into the APK's resources; an edited one under an unchanged name is not
  detected. On iOS the bundle's images and sounds are plain files in the app, so they travel with
  the swap.
* **Use a full build for committed performance captures**, whose provenance names a built commit.
* The tool refuses `art.splotch.app`, an artifact whose bundle id is not `--bundle`, a bundle built
  for a different Hermes bytecode version, and an `--out` that already holds files.
